cmake_minimum_required(VERSION 3.16)
project(nesrecomp1 C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_EXTENSIONS OFF)

# Filled in as the milestones land (see doc/plan.md):
#   runtime/   (M1)   portable C++ core with a C ABI, built as a static library `nesrt`
#   platform/  (M3+)  desktop / web / stm32 front ends
if(EXISTS ${CMAKE_SOURCE_DIR}/runtime/CMakeLists.txt)
  add_subdirectory(runtime)
endif()

include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  # Headless harness: writes NESDUMP 3 files comparable with nes1 and FCEUX (needs the nes1 submodule).
  set(NES1_DIR "${CMAKE_SOURCE_DIR}/deps/nes1" CACHE PATH "nes1 checkout providing common/nesdump.h")
  if(TARGET nrt AND EXISTS ${NES1_DIR}/common/nesdump.h)
    add_executable(nrt_test tests/nrt_test.cpp)
    target_link_libraries(nrt_test PRIVATE nrt)
    target_include_directories(nrt_test PRIVATE ${NES1_DIR})
    target_compile_definitions(nrt_test PRIVATE _CRT_SECURE_NO_WARNINGS)
    if(MSVC)
      target_compile_options(nrt_test PRIVATE /W4 /WX)
    else()
      target_compile_options(nrt_test PRIVATE -Wall -Wextra -Werror)
    endif()
    add_executable(nrt_isolation_test tests/rt_isolation_test.cpp)
    target_link_libraries(nrt_isolation_test PRIVATE nrt)
    target_compile_definitions(nrt_isolation_test PRIVATE _CRT_SECURE_NO_WARNINGS)
    if(MSVC)
      target_compile_options(nrt_isolation_test PRIVATE /W4 /WX)
    else()
      target_compile_options(nrt_isolation_test PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME nrt_isolation COMMAND nrt_isolation_test ${NES1_DIR}/test/probe/controller_serial.nes)
  endif()
  add_executable(nesrt_abi_smoke tests/abi_smoke.c tests/abi_smoke.cpp)
  target_include_directories(nesrt_abi_smoke PRIVATE runtime/include)
  if(MSVC)
    # Flag masks in the generated-operation macros are intentionally compile-time constants.
    target_compile_options(nesrt_abi_smoke PRIVATE /W4 /WX /wd4127)
  else()
    target_compile_options(nesrt_abi_smoke PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME nesrt_abi_smoke COMMAND nesrt_abi_smoke)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME tools_unit COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s tests -p "test_*.py")
    set_tests_properties(tools_unit PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
endif()
